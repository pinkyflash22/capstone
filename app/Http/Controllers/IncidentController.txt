    /**
     * Store a newly created resource in storage.
     */
    public function store(Request $request)
    {
        $request->validate([
            'marker_id' => 'required|exists:markers,marker_id',
            'station_id' => 'nullable|exists:stations,station_id',
            'status' => 'required|in:reported,responding,responded',
            'alarmdatetime' => 'required',
            'reported_by' => 'required|exists:accounts,account_id',
            'incident_type' => 'required|in:fire,drown,treefall,special',
        ]);
        
        $accountId = Auth::user()->account->account_id;
        // $personnels = Personnel::where();

        $incident = Incident::create([
            'marker_id' => $request->marker_id,
            'station_id' => $request->station_id,
            'status' => $request->status,
            'alarmdatetime' => Carbon::now(),
            'reported_by' => $accountId,
            'incident_type' => $request->incident_type,
        ]);

        return response()->json([
            'success' => true,
            'incident' => $incident
        ]);

        return redirect('/citizen/dashboard');
    }

    /**

        $mapmarker = Marker::with(['coordinate.image', 'category'])
            ->whereHas('coordinate')
            ->get();

        ));
    }
    
    /**
     * Get incident chart data with flexible date range
     */
    private function getIncidentChartData($days = 7, $startDate = null, $endDate = null)
    {
        $chartLabels = [];
        $chartValues = [];
        
        // Case 1: Custom date range provided
        if ($startDate && $endDate) {
            $start = Carbon::parse($startDate)->startOfDay();
            $end = Carbon::parse($endDate)->endOfDay();
            $interval = $start->diffInDays($end);
            
            // If range is more than 30 days, group by weeks
            if ($interval > 30) {
                return $this->getWeeklyChartData($start, $end);
            }
            
            // Otherwise group by days
            $current = clone $start;
            while ($current <= $end) {
                $chartLabels[] = $current->format('M d');
                $chartValues[] = Incident::whereDate('created_at', $current)->count();
                $current->addDay();
            }
        }
        // Case 2: Predefined days (7, 30, 90, etc.)
        else {
            $days = (int) $days;
            
            // If more than 30 days, group by weeks
            if ($days > 30) {
                return $this->getWeeklyChartData(Carbon::now()->subDays($days), Carbon::now());
            }
            
            // Group by days
            for ($i = $days - 1; $i >= 0; $i--) {
                $date = Carbon::now()->subDays($i);
                $chartLabels[] = $date->format('D');
                $chartValues[] = Incident::whereDate('created_at', $date)->count();
            }
        }
        
        return [
            'labels' => $chartLabels,
            'values' => $chartValues,
            'type' => 'daily'
        ];
    }
    
    /**
     * Get weekly grouped chart data for longer periods
     */
    private function getWeeklyChartData($startDate, $endDate)
    {
        $chartLabels = [];
        $chartValues = [];
        
        $current = clone $startDate;
        $weekNumber = 1;
        
        while ($current <= $endDate) {
            $weekEnd = (clone $current)->addDays(6);
            if ($weekEnd > $endDate) {
                $weekEnd = clone $endDate;
            }
            
            $chartLabels[] = 'Week ' . $weekNumber . ' (' . $current->format('M d') . ' - ' . $weekEnd->format('M d') . ')';
            $chartValues[] = Incident::whereBetween('created_at', [$current, $weekEnd])->count();
            
            $current->addDays(7);
            $weekNumber++;
        }
        
        return [
            'labels' => $chartLabels,
            'values' => $chartValues,
            'type' => 'weekly'
        ];
    }
    
    /**
     * API endpoint to get chart data (for AJAX requests)
     */
    public function getChartData(Request $request)
    {
        $period = $request->get('period', 'week'); // day, week, month, year, custom
        $startDate = $request->get('start_date');
        $endDate = $request->get('end_date');
        $customDays = $request->get('days');
        
        switch ($period) {
            case 'day':
                return response()->json($this->getHourlyChartData());
                
            case 'week':
                return response()->json($this->getIncidentChartData(7));
                
            case 'month':
                return response()->json($this->getIncidentChartData(30));
                
            case 'quarter':
                return response()->json($this->getIncidentChartData(90));
                
            case 'year':
                return response()->json($this->getMonthlyChartData());
                
            case 'custom':
                if ($customDays) {
                    return response()->json($this->getIncidentChartData($customDays));
                }
                return response()->json($this->getIncidentChartData(30, $startDate, $endDate));
                
            default:
                return response()->json($this->getIncidentChartData(7));
        }
    }
    
    /**
     * Get hourly chart data for today
     */
    private function getHourlyChartData()
    {
        $labels = [];
        $values = [];
        
        for ($hour = 0; $hour < 24; $hour++) {
            $labels[] = sprintf('%02d:00', $hour);
            $values[] = Incident::whereDate('created_at', Carbon::today())
                ->whereTime('created_at', '>=', sprintf('%02d:00:00', $hour))
                ->whereTime('created_at', '<', sprintf('%02d:00:00', ($hour + 1) % 24))
                ->count();
        }
        
        return [
            'labels' => $labels,
            'values' => $values,
            'type' => 'hourly'
        ];
    }
    
    /**
     * Get monthly chart data for the year
     */
    private function getMonthlyChartData()
    {
        $labels = [];
        $values = [];
        
        for ($month = 1; $month <= 12; $month++) {
            $labels[] = Carbon::create()->month($month)->format('M');
            $values[] = Incident::whereMonth('created_at', $month)
                ->whereYear('created_at', Carbon::now()->year)
                ->count();
        }
        
        return [
            'labels' => $labels,
            'values' => $values,
            'type' => 'monthly'
        ];
    }
    
    /**
     * Get comparative statistics
     */
    public function getStatistics()
    {
        $today = Incident::whereDate('created_at', Carbon::today())->count();
        $yesterday = Incident::whereDate('created_at', Carbon::yesterday())->count();
        $thisWeek = Incident::whereBetween('created_at', [Carbon::now()->startOfWeek(), Carbon::now()])->count();
        $lastWeek = Incident::whereBetween('created_at', [Carbon::now()->subWeek()->startOfWeek(), Carbon::now()->subWeek()->endOfWeek()])->count();
        $thisMonth = Incident::whereMonth('created_at', Carbon::now()->month)->count();
        $lastMonth = Incident::whereMonth('created_at', Carbon::now()->subMonth()->month)->count();
        
        $byStatus = [
            'reported' => Incident::where('status', 'reported')->count(),
            'responding' => Incident::where('status', 'responding')->count(),
            'completed' => Incident::where('status', 'completed')->count(),
        ];
        
        $bySeverity = [
            'low' => Incident::where('severity', 'low')->count(),
            'medium' => Incident::where('severity', 'medium')->count(),
            'high' => Incident::where('severity', 'high')->count(),
            'critical' => Incident::where('severity', 'critical')->count(),
        ];
        
        return response()->json([
            'today' => $today,
            'yesterday' => $yesterday,
            'change_today' => $this->calculatePercentageChange($today, $yesterday),
            'this_week' => $thisWeek,
            'last_week' => $lastWeek,
            'change_week' => $this->calculatePercentageChange($thisWeek, $lastWeek),
            'this_month' => $thisMonth,
            'last_month' => $lastMonth,
            'change_month' => $this->calculatePercentageChange($thisMonth, $lastMonth),
            'by_status' => $byStatus,
            'by_severity' => $bySeverity,
        ]);
    }
    
    /**
     * Calculate percentage change between two values
     */
    private function calculatePercentageChange($current, $previous)
    {
        if ($previous == 0) {
            return $current > 0 ? 100 : 0;
        }
        return round((($current - $previous) / $previous) * 100, 1);
    }
    
    /**
     * Get incidents filtered by various criteria
     */
    public function getFilteredIncidents(Request $request)
    {
        $query = Incident::with('marker');
        
        // Filter by date range
        if ($request->has('start_date') && $request->has('end_date')) {
            $query->whereBetween('created_at', [
                Carbon::parse($request->start_date)->startOfDay(),
                Carbon::parse($request->end_date)->endOfDay()
            ]);
        }
        
        // Filter by status
        if ($request->has('status') && $request->status != 'all') {
            $query->where('status', $request->status);
        }
        
        // Filter by incident type
        if ($request->has('incident_type') && $request->incident_type != 'all') {
            $query->where('incident_type', $request->incident_type);
        }
        
        // Filter by station
        if ($request->has('station_id') && $request->station_id != 'all') {
            $query->where('station_id', $request->station_id);
        }
        
        $incidents = $query->orderBy('created_at', 'desc')->get();
        
        return response()->json($incidents);
    }
    
    /**
     * Export incident data
     */
    public function exportData(Request $request)
    {
        $format = $request->get('format', 'csv');
        $startDate = $request->get('start_date', Carbon::now()->subDays(30));
        $endDate = $request->get('end_date', Carbon::now());
        
        $incidents = Incident::with('marker', 'station')
            ->whereBetween('created_at', [$startDate, $endDate])
            ->orderBy('created_at', 'desc')
            ->get();
        
        if ($format == 'csv') {
            $filename = 'incidents_' . Carbon::now()->format('Y-m-d') . '.csv';
            $handle = fopen('php://output', 'w');
            
            header('Content-Type: text/csv');
            header('Content-Disposition: attachment; filename="' . $filename . '"');
            
            // Add headers
            fputcsv($handle, ['ID', 'Title', 'Location', 'Status', 'Type', 'Date', 'Station']);
            
            // Add data
            foreach ($incidents as $incident) {
                fputcsv($handle, [
                    $incident->incident_id,
                    $incident->marker->title ?? 'N/A',
                    ($incident->marker->lat ?? 'N/A') . ', ' . ($incident->marker->lng ?? 'N/A'),
                    $incident->status,
                    $incident->incident_type,
                    $incident->created_at->format('Y-m-d H:i'),
                    $incident->station->station_name ?? 'N/A'
                ]);
            }
            
            fclose($handle);
            exit;
        }
        
        return response()->json($incidents);
    }
