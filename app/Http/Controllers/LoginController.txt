public function authenticate(Request $request) : RedirectResponse
{
    $credentials = $request->validate([
        'email' => 'required|email',
        'password' => 'required'
    ]);

    if (Auth::attempt($credentials)) {
        $request->session()->regenerate();
        
        // Add cache control headers to the response
        $response = redirect()->intended($this->redirectBasedOnRole());
        
        return $response->withHeaders([
            'Cache-Control' => 'no-cache, no-store, must-revalidate, max-age=0',
            'Pragma' => 'no-cache',
            'Expires' => 'Fri, 01 Jan 1990 00:00:00 GMT',
        ]);
    }

    return back()->withErrors([
        'email' => 'The provided credentials do not match our records.',
    ])->onlyInput('email');
}

private function redirectBasedOnRole()
{
    $user = Auth::user();
    $userRole = $user->userInfo?->role?->role;
    
    switch ($userRole) {
        case 'ADMIN':
            return route('admin.dashboard');
        case 'FIREFIGHTER':
            return route('firefighter.dashboard');
        case 'CITIZEN':
            return route('citizen.dashboard');
        default:
            return route('citizen.dashboard');
    }
}

public function logout(Request $request): RedirectResponse
{
    Auth::logout();
    
    $request->session()->invalidate();
    $request->session()->regenerateToken();
    
    // Clear all session data
    $request->session()->flush();
    
    // Redirect with cache control headers
    return redirect('/')
        ->withHeaders([
            'Cache-Control' => 'no-cache, no-store, must-revalidate, max-age=0',
            'Pragma' => 'no-cache',
            'Expires' => 'Fri, 01 Jan 1990 00:00:00 GMT',
        ]);
}
    /**
     * Get user role from the relationship
     */
    private function getUserRole($user)
    {
        // Method 1: If you have the relationships set up
        if ($user->account && $user->account->info && $user->account->info->role) {
            return $user->account->info->role->role;
        }
        
        // Method 2: Direct query if relationships aren't set up
        $account = Account::where('user_id', $user->id)->first();
        if ($account) {
            $userInfo = UserInfo::where('user_id', $account->info_id)->first();
            if ($userInfo) {
                // You might need to get the actual role name from roles table
                // This depends on your exact database structure
                return $userInfo->role_id; // or get role name from roles table
            }
        }
        
        // Default role if nothing found
        return 'CITIZEN';
    }
